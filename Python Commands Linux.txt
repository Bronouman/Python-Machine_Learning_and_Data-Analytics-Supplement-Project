# 1. Start the headless Wayland compositor if it isn't running
export WLR_BACKENDS=headless
export WLR_LIBINPUT_NO_DEVICES=1
labwc &
sleep 2

# 2. Start the Wayland VNC proxy server
export WAYLAND_DISPLAY=wayland-0
wayvnc 127.0.0.1 5900 &
sleep 1

# 3. Execute your Python data analysis script targetting the headless display
WAYLAND_DISPLAY=wayland-0 python analyses_script.py
